import { handleToothClick, selectComponent } from "../src/js/2D/manager.js";
import { createInitialState } from "../src/js/2D/state.js";

function createMissingToothState(toothId = 11) {
  const state = createInitialState();
  state.teeth[toothId].presence = "missing";
  return state;
}

test("clicking the same mesh component toggles it off", () => {
  const state = createMissingToothState(11);
  selectComponent(state, "mesh-hole");

  handleToothClick(state, 11);
  expect(state.teeth[11].components).toEqual(["mesh-hole"]);

  handleToothClick(state, 11);
  expect(state.teeth[11].components).toEqual([]);
  expect(state.statusText).toBe("Removed Mesh Hole from tooth 11.");
});

test("placing a different mesh replaces the existing mesh on the same tooth", () => {
  const state = createMissingToothState(11);

  selectComponent(state, "mesh-hole");
  handleToothClick(state, 11);
  expect(state.teeth[11].components).toEqual(["mesh-hole"]);

  selectComponent(state, "mesh-cross");
  handleToothClick(state, 11);

  expect(state.teeth[11].components).toEqual(["mesh-cross"]);
  expect(state.statusText).toBe("Replaced conflict and placed Mesh Cross on tooth 11.");
});
