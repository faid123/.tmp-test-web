function normalizeBarPlacementOffset(offset) {
  return {
    x: Number.isFinite(offset?.x) ? offset.x : 0,
    y: Number.isFinite(offset?.y) ? offset.y : 0,
    rotation: Number.isFinite(offset?.rotation) ? offset.rotation : 0,
  };
}

function toToothNumber(toothId) {
  const n = Number(toothId);
  return Number.isFinite(n) ? n : null;
}

function normalizeBarSurface(surface) {
  const surfaceKey = String(surface || BAR_DEFAULT_SURFACE).toLowerCase();
  return surfaceKey === "mesial" || surfaceKey === "distal"
    ? surfaceKey
    : BAR_DEFAULT_SURFACE;
}

export function getBarTemplateToothId(toothId) {
  const n = toToothNumber(toothId);
  if (n === null) return String(toothId ?? "");

  const unit = n % 10;
  const quadrant = Math.floor(n / 10);
  if (quadrant === 2) return `1${unit}`;
  if (quadrant === 3) return `4${unit}`;
  return String(n);
}

export function isBarComponent(componentOrId) {
  if (typeof componentOrId === "string") {
    return componentOrId.startsWith("bar-");
  }
  return componentOrId?.tab === BAR_DEFAULT_TAB_ID || String(componentOrId?.id || "").startsWith("bar-");
}

export function getBarPlacementAssetReference(componentId, toothId, surface) {
  const templateToothId = getBarTemplateToothId(toothId);
  const stem = BAR_ASSET_STEM_BY_COMPONENT_ID[componentId] || BAR_ASSET_STEM_BY_COMPONENT_ID[BAR_DEFAULT_COMPONENT_ID];
  if (!stem) return null;

  const normalizedSurface = normalizeBarSurface(surface);
  const supportedSurfaces = BAR_AVAILABLE_SURFACES_BY_TEMPLATE_TOOTH[templateToothId];
  if (!supportedSurfaces || !supportedSurfaces.includes(normalizedSurface)) {
    return null;
  }

  const suffix = `${stem}_${normalizedSurface}`;

  return `${BAR_ASSET_BASE}/${templateToothId}/bars/${templateToothId}_${suffix}.svg`;
}

export function getBarPlacementImageSize(componentId, toothId, surface, token) {

  void token;
  return BAR_IMAGE_SIZE;
}

export function getBarPlacementRenderScale(componentId, toothId, surface, jaw, token) {
  void componentId;
  void jaw;
  void token;

  const toothKey = String(toothId ?? "");
  const templateKey = getBarTemplateToothId(toothId);
  const surfaceKey = String(surface || BAR_DEFAULT_SURFACE).toLowerCase();
  const fallbackSurfaceKey = surfaceKey === BAR_DEFAULT_SURFACE ? BAR_LEGACY_SURFACE_FALLBACK : BAR_DEFAULT_SURFACE;
  const override =
    BAR_RENDER_SCALE_BY_TOOTH_SURFACE[toothKey]?.[surfaceKey] ??
    BAR_RENDER_SCALE_BY_TOOTH_SURFACE[toothKey]?.[fallbackSurfaceKey] ??
    BAR_RENDER_SCALE_BY_TOOTH_SURFACE[templateKey]?.[surfaceKey] ??
    BAR_RENDER_SCALE_BY_TOOTH_SURFACE[templateKey]?.[fallbackSurfaceKey];
  return Number.isFinite(override) ? override : BAR_DEFAULT_RENDER_SCALE;
}

export function getBarPlacementOffset(componentId, toothId, surface, jaw, token) {
  void componentId;
  void jaw;
  void token;

  const toothKey = String(toothId ?? "");
  const templateKey = getBarTemplateToothId(toothId);
  const surfaceKey = String(surface || BAR_DEFAULT_SURFACE).toLowerCase();
  const fallbackSurfaceKey = surfaceKey === BAR_DEFAULT_SURFACE ? BAR_LEGACY_SURFACE_FALLBACK : BAR_DEFAULT_SURFACE;
  const override =
    BAR_PLACEMENT_OFFSET_BY_TOOTH_SURFACE[toothKey]?.[surfaceKey] ??
    BAR_PLACEMENT_OFFSET_BY_TOOTH_SURFACE[toothKey]?.[fallbackSurfaceKey] ??
    BAR_PLACEMENT_OFFSET_BY_TOOTH_SURFACE[templateKey]?.[surfaceKey] ??
    BAR_PLACEMENT_OFFSET_BY_TOOTH_SURFACE[templateKey]?.[fallbackSurfaceKey];
  return normalizeBarPlacementOffset(override);
}

export function getBarMirrorScaleX(toothId, jawFlipX = 1) {
  const n = toToothNumber(toothId);
  if (n === null) return jawFlipX;

  const quadrant = Math.floor(n / 10);
  const mirrored = quadrant === 2 || quadrant === 3;
  return (mirrored ? -1 : 1) * jawFlipX;
}
