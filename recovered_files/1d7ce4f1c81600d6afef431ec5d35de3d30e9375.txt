function buildMeshImageSizeOverrideByTooth(seedByTooth) {
  const result = {};

  for (const toothId of MESH_TOOTH_IDS) {
    const byComponent = {};

    for (const componentId of MESH_COMPONENT_IDS) {
      const baseSize = MESH_IMAGE_SIZE_BY_COMPONENT[componentId] || { width: 154, height: 246 };
      const override = seedByTooth[toothId]?.[componentId] || {};
      byComponent[componentId] = {
        width: Number.isFinite(override.width) ? override.width : baseSize.width,
        height: Number.isFinite(override.height) ? override.height : baseSize.height,
      };
    }

    result[toothId] = byComponent;
  }

  return Object.freeze(result);
}

function buildMeshPositionOffsetByTooth(seedByTooth) {
  const result = {};

  for (const toothId of MESH_TOOTH_IDS) {
    const byComponent = {};

    for (const componentId of MESH_COMPONENT_IDS) {
      const baseOffset = MESH_POSITION_OFFSET_BY_COMPONENT[componentId] || { x: 0, y: 0 };
      const override = seedByTooth[toothId]?.[componentId] || {};
      byComponent[componentId] = {
        x: Number.isFinite(override.x) ? override.x : baseOffset.x,
        y: Number.isFinite(override.y) ? override.y : baseOffset.y,
      };
    }

    result[toothId] = byComponent;
  }

  return Object.freeze(result);
}

function isMeshComponentId(componentId) {
  return String(componentId || "").startsWith("mesh-");
}

function getMeshOverrideValue(overrideMap, toothId, componentId) {
  const exactToothId = String(toothId ?? "");
  const templateToothId = getComponentTemplateToothId(toothId);

  const exactOverride = overrideMap?.[exactToothId]?.[componentId];
  if (exactOverride !== undefined) {
    return exactOverride;
  }

  return overrideMap?.[templateToothId]?.[componentId];
}

export function getMeshPlacementRenderScale(componentId, toothId, jaw) {
  const componentScale = MESH_RENDER_SCALE_BY_COMPONENT[componentId] || 1;
  const jawScale = MESH_RENDER_SCALE_BY_JAW[jaw] || 1;
  const override = getMeshOverrideValue(MESH_RENDER_SCALE_OVERRIDE_BY_TOOTH, toothId, componentId);

  if (Number.isFinite(override)) {
    return override;
  }

  return componentScale * jawScale;
}

export function getMeshPlacementOffset(componentId, toothId) {
  const base = MESH_POSITION_OFFSET_BY_COMPONENT[componentId] || { x: 0, y: 0 };
  const override = getMeshOverrideValue(MESH_POSITION_OFFSET_BY_TOOTH, toothId, componentId) || {};

  const x = Number.isFinite(override.x) ? override.x : base.x;
  const y = Number.isFinite(override.y) ? override.y : base.y;
  return { x, y };
}

export function getMeshPlacementImageSize(componentId, toothId) {
  const base = MESH_IMAGE_SIZE_BY_COMPONENT[componentId] || null;
  const override = getMeshOverrideValue(MESH_IMAGE_SIZE_OVERRIDE_BY_TOOTH, toothId, componentId) || null;

  if (override && Number.isFinite(override.width) && Number.isFinite(override.height)) {
    return { width: override.width, height: override.height };
  }

  if (base && Number.isFinite(base.width) && Number.isFinite(base.height)) {
    return { width: base.width, height: base.height };
  }

  return null;
}

// Normalize tooth id to the uploaded template tooth id used for component assets.
export function getComponentTemplateToothId(toothId) {
  const numeric = Number(toothId);
  if (!Number.isFinite(numeric)) {
    return "11";
  }

  const unit = numeric % 10;
  const quadrant = Math.floor(numeric / 10);
  if (quadrant === 2) {
    return `1${unit}`;
  }
  if (quadrant === 3) {
    return `4${unit}`;
  }
  return `${quadrant}${unit}`;
}

// Resolve component image path by component id and target tooth unit.
export function getComponentAssetReference(componentId, toothId) {
  const templateToothId = getComponentTemplateToothId(toothId);
  const suffix = COMPONENT_IMAGE_SUFFIX_BY_ID[componentId];
  if (!suffix) {
    return null;
  }

  if (isMeshComponentId(componentId)) {
    return `${COMPONENT_ASSET_BASE}/${templateToothId}/mesh/${templateToothId}-${suffix}.svg`;
  }

  const fileName = `${templateToothId}-${suffix}`;
  if (!fileName) {
    return null;
  }
  return `${COMPONENT_ASSET_BASE}/${templateToothId}/${fileName}`;
}

function getToothUnit(toothId) {
  const numeric = Number(toothId);
  if (!Number.isFinite(numeric)) {
    return null;
  }
  return numeric % 10;
}

function getRestVariantForTooth(toothId) {
  const unit = getToothUnit(toothId);
  if (!Number.isFinite(unit)) {
    return "posterior";
  }
  return unit >= 1 && unit <= 3 ? "anterior" : "posterior";
}

function normalizeRestSurface(surface) {
  if (typeof surface !== "string") {
    return null;
  }

  const normalized = surface.toLowerCase();
  return normalized === "occlusal" ? "lingual" : normalized;
}

function getRestPlacementToken(componentId, toothId, surface) {
  const variant = getRestVariantForTooth(toothId);
  const normalizedSurface = normalizeRestSurface(surface);
  if (!normalizedSurface) {
    return null;
  }

  if (variant === "anterior") {
    void componentId;
    if (normalizedSurface === "mesial") return "ai_mesial";
    if (normalizedSurface === "distal") return "ai_distal";
    if (normalizedSurface === "lingual") {
      return "ac_full";
    }
    return null;
  }

  if (normalizedSurface === "mesial") return "p_mesial";
  if (normalizedSurface === "distal") return "p_distal";
  if (normalizedSurface === "lingual") return "p_lingual";
  return null;
}

function getRestOverrideValue(overrideMap, toothId, token) {
  const exactToothId = String(toothId ?? "");
  const exactOverride = overrideMap?.[exactToothId]?.[token];
  if (exactOverride !== undefined) {
    return exactOverride;
  }

  const templateToothId = getComponentTemplateToothId(toothId);
  return overrideMap?.[templateToothId]?.[token];
}

export function getRestPlacementAssetReference(componentId, toothId, surface) {
  const templateToothId = getComponentTemplateToothId(toothId);
  const token = getRestPlacementToken(componentId, toothId, surface);

  if (!token) {
    return null;
  }

  if ((templateToothId === "17" || templateToothId === "18") && token === "p_mesial") {
    return `${REST_PLACEMENT_ASSET_BASE}/${templateToothId}/rest/${templateToothId}-p_meisal.svg`;
  }

  return `${REST_PLACEMENT_ASSET_BASE}/${templateToothId}/rest/${templateToothId}-${token}.svg`;
}

export function getRestPlacementImageSize(componentId, toothId, surface) {
  const token = getRestPlacementToken(componentId, toothId, surface);
  if (!token) {
    return null;
  }

  const byTooth = getRestOverrideValue(REST_PLACEMENT_IMAGE_SIZE_BY_TOOTH, toothId, token);
  if (byTooth && Number.isFinite(byTooth.width) && Number.isFinite(byTooth.height)) {
    return byTooth;
  }

  return REST_PLACEMENT_IMAGE_SIZE[token] || null;
}

export function getRestPlacementRenderScale(componentId, toothId, surface, jaw = null) {
  const token = getRestPlacementToken(componentId, toothId, surface);
  if (!token) {
    const jawScale = REST_PLACEMENT_SCALE_BY_JAW[jaw] || 1;
    return 0.55 * jawScale;
  }

  const override = getRestOverrideValue(REST_PLACEMENT_SCALE_OVERRIDE_BY_TOOTH, toothId, token);
  if (Number.isFinite(override)) {
    return override;
  }

  const tokenScale = REST_PLACEMENT_SCALE_BY_TOKEN[token] || 0.55;
  const jawScale = REST_PLACEMENT_SCALE_BY_JAW[jaw] || 1;
  return tokenScale * jawScale;
}

export function getRestSuggestionSurfaces(componentId, toothId) {
  const variant = getRestVariantForTooth(toothId);
  if (variant === "anterior" && componentId === "rest-onlay") {
    return [REST_SURFACE.MESIAL, REST_SURFACE.DISTAL];
  }
  return [...REST_SURFACE_ORDER];
}

export function getRestSuggestionRadius() {
  return REST_SUGGESTION_CONFIG.radius;
}

export function isRestComponent(componentOrId) {
  const def =
    typeof componentOrId === "string"
      ? COMPONENT_BY_ID.get(componentOrId)
      : componentOrId;
  return def?.tab === "rests";
}

export function getRestSuggestionPointsForTooth(
  toothId,
  jaw,
  mirrored,
  jawFlipX = 1,
  componentId = null
) {
  void jaw;
  const direction = (mirrored ? -1 : 1) * jawFlipX;
  const override = REST_SUGGESTION_POINT_OVERRIDES[toothId] || {};
  const allowedSurfaces = new Set(getRestSuggestionSurfaces(componentId, toothId));

  return REST_SURFACE_ORDER.map((surface) => {
    const base = REST_SUGGESTION_CONFIG.defaultPoints[surface];
    const surfaceOverride = override[surface] || {};
    const x = surface === REST_SURFACE.LINGUAL ? base.x : base.x * direction;

    return {
      surface,
      x: Number.isFinite(surfaceOverride.x) ? surfaceOverride.x : x,
      y: Number.isFinite(surfaceOverride.y) ? surfaceOverride.y : base.y,
    };
  }).filter((point) => allowedSurfaces.has(point.surface));
}
