import { ACTION_UPON_FAILURE } from "./components.js";
import { JAW, MODE } from "./constants.js";
import { getComponentById } from "./components-catalog.js";
import { assessPlacement } from "./criteria.js";
import { applyCommand, pushHistory, redo, undo } from "./history.js";
import { setStatus } from "./state.js";

function createCommand(label, deltas) {
  return {
    label,
    deltas,
  };
}

function normalizeDelta(delta) {
  return {
    toothId: delta.toothId,
    add: [...new Set(delta.add || [])],
    remove: [...new Set(delta.remove || [])],
    presenceBefore: delta.presenceBefore,
    presenceAfter: delta.presenceAfter,
  };
}

export function selectComponent(state, componentId) {
  state.selectedComponentId = componentId;
  const component = getComponentById(componentId);
  setStatus(state, component ? `Selected: ${component.label}` : "No component selected.");
}

export function setMode(state, mode) {
  state.mode = mode;
  setStatus(state, mode === MODE.PRESENCE ? "Presence mode active." : "Design mode active.");
}

export function handleToothClick(state, toothId) {
  const tooth = state.teeth[toothId];
  if (!tooth) {
    return;
  }

  if (state.mode === MODE.PRESENCE) {
    togglePresence(state, toothId);
    return;
  }

  placeSelectedComponent(state, toothId);
}

function togglePresence(state, toothId) {
  const tooth = state.teeth[toothId];
  const nextPresence = tooth.presence === "present" ? "missing" : "present";

  const toRemove = [];
  if (nextPresence === "missing") {
    for (const compId of tooth.components) {
      const definition = getComponentById(compId);
      if (!definition) {
        continue;
      }

      if (definition.criteria.includes("ToothPresent")) {
        toRemove.push(compId);
      }
    }
  }

  const command = createCommand(`Toggle presence on ${toothId}`, [
    normalizeDelta({
      toothId,
      remove: toRemove,
      add: [],
      presenceBefore: tooth.presence,
      presenceAfter: nextPresence,
    }),
  ]);

  applyCommand(state, command, "forward");
  pushHistory(state, command);
  setStatus(state, `Tooth ${toothId} is now ${nextPresence}.`);
}

function placeSelectedComponent(state, toothId) {
  if (!state.selectedComponentId) {
    setStatus(state, "Select a component first.");
    return;
  }

  const tooth = state.teeth[toothId];
  const componentDef = getComponentById(state.selectedComponentId);
  if (!componentDef) {
    setStatus(state, "Invalid component selection.");
    return;
  }

  if (tooth.components.includes(componentDef.id)) {
    const command = createCommand(`Remove ${componentDef.label}`, [
      normalizeDelta({ toothId, add: [], remove: [componentDef.id] }),
    ]);
    applyCommand(state, command, "forward");
    pushHistory(state, command);
    setStatus(state, `Removed ${componentDef.label} from tooth ${toothId}.`);
    return;
  }

  const assessment = assessPlacement(tooth, componentDef);
  if (assessment.pass) {
    const command = createCommand(`Place ${componentDef.label}`, [normalizeDelta({ toothId, add: [componentDef.id], remove: [] })]);
    applyCommand(state, command, "forward");
    pushHistory(state, command);
    setStatus(state, `Placed ${componentDef.label} on tooth ${toothId}.`);
    return;
  }

  const { failureData } = assessment;
  if (failureData.actionUponFailure === ACTION_UPON_FAILURE.PREVENT_PLACEMENT) {
    setStatus(state, failureData.reason);
    return;
  }

  if (failureData.actionUponFailure === ACTION_UPON_FAILURE.REMOVE_THEN_PLACE) {
    const command = createCommand(`Replace conflict with ${componentDef.label}`, [
      normalizeDelta({
        toothId,
        remove: failureData.conflictingComponents || [],
        add: [componentDef.id],
      }),
    ]);
    applyCommand(state, command, "forward");
    pushHistory(state, command);
    setStatus(state, `Replaced conflict and placed ${componentDef.label} on tooth ${toothId}.`);
  }
}

export function clearJaw(state, jaw) {
  const ids = jaw === JAW.UPPER ? [18, 17, 16, 15, 14, 13, 12, 11, 21, 22, 23, 24, 25, 26, 27, 28] : [38, 37, 36, 35, 34, 33, 32, 31, 41, 42, 43, 44, 45, 46, 47, 48];
  const deltas = [];

  for (const id of ids) {
    const tooth = state.teeth[id];
    if (!tooth.components.length) {
      continue;
    }

    deltas.push(normalizeDelta({ toothId: id, remove: [...tooth.components], add: [] }));
  }

  if (!deltas.length) {
    setStatus(state, `No components to clear on ${jaw} jaw.`);
    return;
  }

  const command = createCommand(`Clear ${jaw} jaw`, deltas);
  applyCommand(state, command, "forward");
  pushHistory(state, command);
  setStatus(state, `Cleared ${jaw} jaw components.`);
}

export function clearAll(state) {
  const deltas = [];

  for (const tooth of Object.values(state.teeth)) {
    if (!tooth.components.length) {
      continue;
    }

    deltas.push(normalizeDelta({ toothId: tooth.fdiId, remove: [...tooth.components], add: [] }));
  }

  if (!deltas.length) {
    setStatus(state, "No components to clear.");
    return;
  }

  const command = createCommand("Clear all", deltas);
  applyCommand(state, command, "forward");
  pushHistory(state, command);
  setStatus(state, "Cleared all components.");
}

export function undoAction(state) {
  if (!undo(state)) {
    setStatus(state, "Nothing to undo.");
    return;
  }

  setStatus(state, "Undo complete.");
}

export function redoAction(state) {
  if (!redo(state)) {
    setStatus(state, "Nothing to redo.");
    return;
  }

  setStatus(state, "Redo complete.");
}
